<stdlib.h>   //Standard Library งานทั่วไป แปลงข้อมูล จัดการหน่วยความจำ ตัวเลขสุ่ม
malloc()     //จองพื้นที่หน่วยความจำ
calloc()     //จองพื้นที่หน่วยความจำและกำหนดค่าเริ่มต้นเป็น 0
realloc()    //ปรับขนาดพื้นที่หน่วยความจำที่จองไว้
free()       //คืนพื้นที่หน่วยความจำ
rand()       //สร้างเลขสุ่ม

<string.h>   //String Handling จัดการข้อความ
strlen()     //หาความยาวของข้อความ
strcpy()     //คัดลอกข้อความ
strcat()     //ต่อข้อความเข้าด้วยกัน
strcmp()     //เปรียบเทียบข้อความ 2 ชุด
strchr()     //ค้นหาตำแหน่งของตัวอักษรในข้อความ

<math.h>     //Mathematical Functions คำนวณทางคณิตศาสตร์
sqrt()       //หาค่ารากที่สอง
pow()        //ยกกำลัง
ceil()       //ปัดค่าขึ้น
floor()      //ปัดค่าลง
fabs()       //หาค่าสัมบูรณ์ของจำนวนชนิด double
